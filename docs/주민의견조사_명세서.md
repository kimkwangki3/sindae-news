# 2. 데이터 모델

## 2-1. 테이블

```sql
-- ─────────────────────────────────────────────
-- 설문
-- ─────────────────────────────────────────────
create table public.surveys (
  id              uuid primary key default gen_random_uuid(),
  slug            text unique not null,
  title           text not null,
  description     text,
  status          text not null default 'draft'
                    check (status in ('draft','open','closed')),
  starts_at       timestamptz,
  ends_at         timestamptz,
  allow_multiple  boolean not null default false,  -- 복수 선택 허용 여부
  max_choices     int not null default 1,          -- 복수 선택 시 최대 개수
  total_votes     int not null default 0,          -- 비정규화 집계(트리거로 갱신)
  created_by      uuid references auth.users(id),
  created_at      timestamptz not null default now(),
  updated_at      timestamptz not null default now(),

  constraint valid_period check (ends_at is null or starts_at is null or ends_at > starts_at),
  constraint valid_max_choices check (max_choices >= 1)
);

create index surveys_status_idx on public.surveys(status, ends_at desc);
create index surveys_slug_idx   on public.surveys(slug);

-- ─────────────────────────────────────────────
-- 보기
-- ─────────────────────────────────────────────
create table public.survey_options (
  id          uuid primary key default gen_random_uuid(),
  survey_id   uuid not null references public.surveys(id) on delete cascade,
  label       text not null,
  sort_order  int  not null default 0,
  vote_count  int  not null default 0,             -- 비정규화 집계
  created_at  timestamptz not null default now()
);

create index survey_options_survey_idx on public.survey_options(survey_id, sort_order);

-- ─────────────────────────────────────────────
-- 투표  ★ 1인 1설문 1표의 최종 방어선
-- ─────────────────────────────────────────────
create table public.survey_votes (
  id          uuid primary key default gen_random_uuid(),
  survey_id   uuid not null references public.surveys(id) on delete cascade,
  option_id   uuid not null references public.survey_options(id) on delete cascade,
  user_id     uuid not null references auth.users(id) on delete cascade,
  district    text,                                 -- 선택: 신대/선월/복성/기타
  age_band    text,                                 -- 선택: 20대/30대/…
  ip_hash     text,                                 -- 원본 저장 금지. 해시만
  created_at  timestamptz not null default now()
);

-- ★★★ 단일 선택 설문의 1인 1표 보장 (DB 레벨)
create unique index survey_votes_one_per_user
  on public.survey_votes(survey_id, user_id);

create index survey_votes_option_idx on public.survey_votes(option_id);
```

> **복수 선택을 허용할 경우** 위 유니크 인덱스를 `(survey_id, user_id, option_id)`로 바꾸고,
> 최대 개수 제한은 3-2의 RPC 함수 안에서 검증한다.
> **1차 버전은 단일 선택만 구현할 것을 권한다.** 조작 방지가 훨씬 단순해진다.

# 3. 핵심 로직 — 투표는 반드시 DB 함수로

## 3-1. 왜 RPC 함수인가

클라이언트에서 `insert`를 직접 호출하면 아래를 막을 수 없다.

- 종료된 설문에 투표
- 다른 설문의 option_id를 섞어 넣기
- 존재하지 않는 option_id
- 집계 카운트 불일치

**모든 검증을 하나의 트랜잭션 안에서 처리하는 DB 함수로 만든다.**

## 3-2. 투표 함수

```sql
create or replace function public.cast_survey_vote(
  p_survey_id  uuid,
  p_option_id  uuid,
  p_district   text default null,
  p_age_band   text default null,
  p_ip_hash    text default null
)
returns json
language plpgsql
security definer
set search_path = public
as $$
declare
  v_user_id uuid := auth.uid();
  v_survey  public.surveys%rowtype;
  v_option  public.survey_options%rowtype;
begin
  -- 1) 로그인 확인
  if v_user_id is null then
    return json_build_object('ok', false, 'code', 'UNAUTHENTICATED');
  end if;

  -- 2) 설문 조회 + 행 잠금 (동시 요청 방어)
  select * into v_survey
  from public.surveys
  where id = p_survey_id
  for update;

  if not found then
    return json_build_object('ok', false, 'code', 'SURVEY_NOT_FOUND');
  end if;

  -- 3) 상태·기간 검증 (서버 시각 기준. 클라이언트 시각 신뢰 금지)
  if v_survey.status <> 'open' then
    return json_build_object('ok', false, 'code', 'SURVEY_NOT_OPEN');
  end if;

  if v_survey.starts_at is not null and now() < v_survey.starts_at then
    return json_build_object('ok', false, 'code', 'NOT_STARTED');
  end if;

  if v_survey.ends_at is not null and now() > v_survey.ends_at then
    return json_build_object('ok', false, 'code', 'ALREADY_ENDED');
  end if;

  -- 4) 보기가 이 설문에 속하는지 검증 ★ 중요
  select * into v_option
  from public.survey_options
  where id = p_option_id and survey_id = p_survey_id;

  if not found then
    return json_build_object('ok', false, 'code', 'INVALID_OPTION');
  end if;

  -- 5) 선택값 화이트리스트 검증 (자유 입력 차단)
  if p_district is not null
     and p_district not in ('신대지구','선월지구','복성지구','해룡면 기타','순천시 기타') then
    return json_build_object('ok', false, 'code', 'INVALID_DISTRICT');
  end if;

  if p_age_band is not null
     and p_age_band not in ('10대','20대','30대','40대','50대','60대 이상') then
    return json_build_object('ok', false, 'code', 'INVALID_AGE_BAND');
  end if;

  -- 6) 투표 삽입. 중복은 유니크 인덱스가 막는다
  begin
    insert into public.survey_votes
      (survey_id, option_id, user_id, district, age_band, ip_hash)
    values
      (p_survey_id, p_option_id, v_user_id, p_district, p_age_band, p_ip_hash);
  exception when unique_violation then
    return json_build_object('ok', false, 'code', 'ALREADY_VOTED');
  end;

  -- 7) 집계 갱신 (같은 트랜잭션)
  update public.survey_options
     set vote_count = vote_count + 1
   where id = p_option_id;

  update public.surveys
     set total_votes = total_votes + 1,
         updated_at  = now()
   where id = p_survey_id;

  return json_build_object('ok', true, 'code', 'VOTED');
end;
$$;

revoke all on function public.cast_survey_vote(uuid,uuid,text,text,text) from public;
grant execute on function public.cast_survey_vote(uuid,uuid,text,text,text) to authenticated;
```

**핵심 포인트**
- `security definer` + `set search_path` — 권한 상승 경로 차단
- `for update` — 동시 요청(더블클릭, 스크립트 연타) 직렬화
- **option이 해당 survey 소속인지 검증** — 가장 흔히 빠뜨리는 취약점
- 자유 입력값을 **화이트리스트로 검증** — 임의 문자열 저장 차단
- `anon`에는 실행 권한을 주지 않는다 (`authenticated`만)

## 3-3. 결과 조회 함수

```sql
create or replace function public.get_survey_results(p_survey_id uuid)
returns json
language sql
stable
security definer
set search_path = public
as $$
  select json_build_object(
    'survey_id',   s.id,
    'title',       s.title,
    'status',      s.status,
    'total_votes', s.total_votes,
    'options', (
      select coalesce(json_agg(
        json_build_object(
          'id',    o.id,
          'label', o.label,
          'count', o.vote_count,
          'ratio', case when s.total_votes = 0 then 0
                        else round(o.vote_count::numeric * 100 / s.total_votes, 1) end
        ) order by o.sort_order
      ), '[]'::json)
      from public.survey_options o
      where o.survey_id = s.id
    )
  )
  from public.surveys s
  where s.id = p_survey_id
    and s.status in ('open','closed');
$$;

grant execute on function public.get_survey_results(uuid) to anon, authenticated;
```

> **개별 투표 레코드는 절대 클라이언트에 노출하지 않는다.** 집계만 내보낸다.
> 참여자가 적을 때 개별 레코드가 노출되면 누가 무엇에 투표했는지 추정될 수 있다.

## 3-4. 내 투표 여부 확인

create or replace function public.get_my_vote(p_survey_id uuid)
returns json

as $$

# 4. RLS 정책

**모든 테이블에 RLS를 켠다. 이게 최후의 방어선이다.**

```sql
alter table public.surveys        enable row level security;
alter table public.survey_options enable row level security;
alter table public.survey_votes   enable row level security;

-- ── surveys ────────────────────────────────
-- 공개된 설문만 누구나 읽기
create policy surveys_select_public on public.surveys
  for select using (status in ('open','closed'));

-- 관리자는 전체 읽기
create policy surveys_select_admin on public.surveys
  for select using (public.is_admin());

-- 쓰기는 관리자만
create policy surveys_write_admin on public.surveys
  for all using (public.is_admin()) with check (public.is_admin());

-- ── survey_options ─────────────────────────
create policy options_select_public on public.survey_options
  for select using (
    exists (select 1 from public.surveys s
             where s.id = survey_id and s.status in ('open','closed'))
  );

create policy options_select_admin on public.survey_options
  for select using (public.is_admin());

create policy options_write_admin on public.survey_options
  for all using (public.is_admin()) with check (public.is_admin());

-- ── survey_votes ───────────────────────────
-- ★ 본인 투표만 읽을 수 있다. 남의 투표는 조회 불가
create policy votes_select_own on public.survey_votes
  for select using (user_id = auth.uid());

-- ★ 직접 INSERT 금지. 반드시 RPC 함수를 거치게 한다
--    (security definer 함수는 RLS를 우회하므로 정상 동작)
--    INSERT / UPDATE / DELETE 정책을 아예 만들지 않는다 = 전부 차단

-- 관리자도 개별 투표는 읽지 못하게 한다 (개인정보 최소화)
-- 집계가 필요하면 get_survey_results() 를 쓴다
```

**주의: `votes` 테이블에 INSERT 정책을 만들지 마십시오.** 정책이 없으면 차단되고, RPC 함수만 통과합니다. 그게 설계 의도입니다.

# 5. API / 서버 액션

## 5-1. 투표 처리 (Route Handler)

```ts
// app/api/surveys/[slug]/vote/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { createServerClient } from '@/lib/supabase/server'
import { createHash } from 'crypto'

const DISTRICTS = ['신대지구','선월지구','복성지구','해룡면 기타','순천시 기타'] as const
const AGE_BANDS = ['10대','20대','30대','40대','50대','60대 이상'] as const

export async function POST(
  req: NextRequest,
  { params }: { params: Promise<{ slug: string }> }
) {
  const { slug } = await params
  const supabase = await createServerClient()

  // 1) 세션 확인 — 서버에서
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) {
    return NextResponse.json({ ok: false, code: 'UNAUTHENTICATED' }, { status: 401 })
  }

  // 2) 입력 파싱 + 타입 검증
  let body: unknown
  try { body = await req.json() } catch {
    return NextResponse.json({ ok: false, code: 'BAD_REQUEST' }, { status: 400 })
  }

  const { optionId, district, ageBand } = (body ?? {}) as Record<string, unknown>

  const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i
  if (typeof optionId !== 'string' || !UUID_RE.test(optionId)) {
    return NextResponse.json({ ok: false, code: 'INVALID_OPTION' }, { status: 400 })
  }
  if (district !== undefined && !DISTRICTS.includes(district as never)) {
    return NextResponse.json({ ok: false, code: 'INVALID_DISTRICT' }, { status: 400 })
  }
  if (ageBand !== undefined && !AGE_BANDS.includes(ageBand as never)) {
    return NextResponse.json({ ok: false, code: 'INVALID_AGE_BAND' }, { status: 400 })
  }

  // 3) slug → survey_id (클라이언트가 보낸 survey_id를 믿지 않는다)
  const { data: survey } = await supabase
    .from('surveys').select('id').eq('slug', slug).eq('status','open').single()

  if (!survey) {
    return NextResponse.json({ ok: false, code: 'SURVEY_NOT_OPEN' }, { status: 404 })
  }

  // 4) IP 해시 (원본 저장 금지)
  const ip = req.headers.get('x-forwarded-for')?.split(',')[0]?.trim() ?? ''
  const salt = process.env.VOTE_IP_SALT!            // 서버 전용 환경변수
  const ipHash = ip ? createHash('sha256').update(salt + ip).digest('hex') : null

  // 5) DB 함수 호출 — 모든 검증은 여기서
  const { data, error } = await supabase.rpc('cast_survey_vote', {
    p_survey_id: survey.id,
    p_option_id: optionId,
    p_district:  district ?? null,
    p_age_band:  ageBand ?? null,
    p_ip_hash:   ipHash,
  })

  if (error) {
    console.error('vote error', error)   // 상세 오류는 로그에만
    return NextResponse.json({ ok: false, code: 'SERVER_ERROR' }, { status: 500 })
  }

  const result = data as { ok: boolean; code: string }
  return NextResponse.json(result, { status: result.ok ? 200 : 409 })
}
```

**하지 말아야 할 것**
- 클라이언트가 보낸 `surveyId`를 그대로 쓰기 → slug로 서버에서 조회
- 클라이언트가 보낸 `userId` 신뢰 → 세션에서만 가져오기
- DB 오류 메시지를 클라이언트에 그대로 반환 → 스키마 정보 유출

## 5-2. Rate Limiting

RPC와 유니크 제약이 중복 투표를 막지만, **무차별 요청 자체를 줄여야 한다.**

```ts
// middleware.ts 또는 Route Handler 상단
// Vercel 사용 시 @upstash/ratelimit + Upstash Redis 권장
// 최소한: 동일 user_id 기준 분당 10회 제한
```

간단히 가려면 DB에 요청 로그 테이블을 두고 카운트해도 된다. **없어도 치명적이지는 않으나 있으면 좋다.**

| # | 공격 | 방어 |

| 2 | 더블클릭·연타로 동시 요청 | `for update` 행 잠금 + UNIQUE |
| 3 | 브라우저 개발자도구로 API 직접 호출 | 서버 세션 검증 + RLS + RPC |

| 6 | 클라이언트에서 시간 조작 | 모든 시간 판단은 DB `now()` 기준 |
| 7 | supabase-js로 votes 테이블 직접 INSERT | **INSERT 정책 미생성** = 차단 |

# 7. 다중 계정 문제 — 가장 현실적인 위협

**1인 1표는 '계정 1개당 1표'입니다. 계정을 여러 개 만들면 뚫립니다.**
기술로 100% 막을 수 없으므로 **비용을 올려 실익을 없애는 것**이 목표다.

## 7-1. 권장 조합

```
① 이메일 인증 필수         미인증 계정은 투표 불가
② 소셜 로그인 우선          카카오/네이버 로그인은 계정 생성 비용이 높다
③ 가입 후 대기 시간         가입 24시간 경과 계정만 투표 가능
④ 설문 개설 이전 가입자만    (선택) 설문 시작 후 만든 계정 배제
⑤ ip_hash 모니터링         같은 해시에서 다수 계정 투표 시 관리자 알림
```

**RPC 함수에 아래를 추가한다.**

```sql
  -- 이메일 인증 및 가입 경과 검증
  if not exists (
    select 1 from auth.users u
     where u.id = v_user_id
       and u.email_confirmed_at is not null
       and u.created_at < now() - interval '24 hours'
  ) then
    return json_build_object('ok', false, 'code', 'ACCOUNT_NOT_ELIGIBLE');
  end if;
```

> 안내 문구: **"가입 24시간이 지난 계정만 참여할 수 있습니다."**
> 조작 의도를 사전에 꺾는 효과가 크다.

## 7-2. 이상 징후 감지 (관리자 화면)

```sql
-- 같은 IP 해시에서 3개 이상 계정이 투표한 경우
select ip_hash, count(distinct user_id) as accounts, count(*) as votes
from public.survey_votes
where survey_id = $1 and ip_hash is not null
group by ip_hash
having count(distinct user_id) >= 3
order by accounts desc;
```

**가족·직장 등 정상적인 경우도 있으므로 자동 차단하지 말고 관리자에게 표시만 한다.**

# 10. 최고관리자 상단 Admin 버튼

**설문과 별개 요구사항.**

```tsx
// app/layout.tsx 또는 components/Header.tsx — 반드시 서버 컴포넌트
import { createServerClient } from '@/lib/supabase/server'

export default async function Header() {
  const supabase = await createServerClient()
  const { data: { user } } = await supabase.auth.getUser()

  let isSuperAdmin = false
  if (user) {
    const { data } = await supabase
      .from('profiles').select('role').eq('id', user.id).single()
    isSuperAdmin = data?.role === 'superadmin'
  }

  return (
    <header>
      {/* 기존 헤더 */}
      {isSuperAdmin && (
        <Link href="/admin" className="admin-btn">관리자</Link>
      )}
    </header>
  )
}
```

**보안 주의**

- **클라이언트 상태로 role을 판단하지 마십시오.** 버튼을 숨기는 건 UI일 뿐입니다
- `/admin/*` 경로는 **middleware와 각 페이지에서 이중으로** role을 검증해야 합니다
- 버튼이 안 보여도 URL을 직접 치면 들어갈 수 있다는 전제로 설계합니다

```ts
// middleware.ts
export async function middleware(req: NextRequest) {
  if (req.nextUrl.pathname.startsWith('/admin')) {
    const supabase = createMiddlewareClient(req)
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) return NextResponse.redirect(new URL('/login', req.url))

    const { data } = await supabase
      .from('profiles').select('role').eq('id', user.id).single()

    if (!['admin','superadmin'].includes(data?.role ?? '')) {
      return NextResponse.redirect(new URL('/', req.url))
    }
  }
  return NextResponse.next()
}

export const config = { matcher: ['/admin/:path*'] }
```
